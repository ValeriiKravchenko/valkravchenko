===== П7.51. GITIGNORE: check-ignore, правило ! совпало
--- .gitignore:
*.log$
!important.log$
---
$ git check-ignore -v important.log
.gitignore:2:!important.log	important.log
[exit 0]

$ git check-ignore -v debug.log important.log
.gitignore:1:*.log	debug.log
.gitignore:2:!important.log	important.log
[exit 0]

$ git check-ignore important.log
[exit 1]

$ git check-ignore debug.log
debug.log
[exit 0]

$ git check-ignore debug.log important.log
debug.log
[exit 0]

$ git check-ignore -v --non-matching important.log build/bundle.js
.gitignore:2:!important.log	important.log
::	build/bundle.js
[exit 0]

$ git check-ignore -v build/bundle.js
[exit 1]

--- .gitignore:
!important.log$
---

$ git check-ignore -v important.log
